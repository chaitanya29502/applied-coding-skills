import java.util.*;

class Solution {
    public int leastInterval(char[] tasks, int n) {

        int[] freq = new int[26];

        for (char task : tasks) {
            freq[task - 'A']++;
        }

        // Max heap: task with highest frequency first
        PriorityQueue<Integer> maxHeap =
                new PriorityQueue<>(Collections.reverseOrder());

        for (int count : freq) {
            if (count > 0) {
                maxHeap.offer(count);
            }
        }

        int time = 0;

        while (!maxHeap.isEmpty()) {

            List<Integer> temp = new ArrayList<>();

            // Try to execute n + 1 different tasks
            for (int i = 0; i <= n; i++) {

                if (!maxHeap.isEmpty()) {
                    int count = maxHeap.poll();

                    if (count - 1 > 0) {
                        temp.add(count - 1);
                    }

                    time++;
                }
                else if (temp.isEmpty()) {
                    break;
                }
                else {
                    // CPU is idle
                    time++;
                }
            }

            // Put remaining tasks back
            for (int count : temp) {
                maxHeap.offer(count);
            }
        }

        return time;
    }
}
